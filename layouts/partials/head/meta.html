{{- /* Meta tags for SEO and social sharing. Overrides the theme's default meta partial to provide: - Description (per-page or site default) - OpenGraph tags - Twitter Card tags - Canonical URL - Robots directives - JSON-LD structured data (injected at end of meta block) */ -}} {{- $site := .Site -}} {{- $page := . -}} {{- /* Description: front matter > summary > site params */ -}} {{- $description := "" -}} {{- with $page.Description -}} {{- $description = . -}} {{- else -}} {{- with $page.Summary -}} {{- $description = (. | plainify) -}} {{- else -}} {{- $description = $site.Params.description | default $site.Title -}} {{- end -}} {{- end -}} {{- $description = $description | plainify | chomp | strings.TrimSpace -}} {{- /* Author: front matter > site params > site author */ -}} {{- $author := "" -}} {{- with $page.Params.author -}} {{- $author = . -}} {{- else -}} {{- with $site.Params.author -}} {{- $author = . -}} {{- else -}} {{- $author = $site.Author.name | default $site.Title -}} {{- end -}} {{- end -}} {{- /* Image: front matter > site params */ -}} {{- $image := "" -}} {{- with $page.Params.cover.image -}} {{- $image = . | absURL -}} {{- else -}} {{- with $page.Params.image -}} {{- $image = . | absURL -}} {{- else -}} {{- with $site.Params.image -}} {{- $image = . | absURL -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Canonical URL */ -}} {{- $canonical := $page.Permalink -}} {{- /* Page type for OpenGraph: only single-content pages are "article"; home/section/taxonomy are "website" */ -}} {{- $ogType := "website" -}} {{- if eq $page.Kind "page" -}} {{- $ogType = "article" -}} {{- end -}} {{- /* Locale */ -}} {{- $locale := $site.Language.Lang | default $site.LanguageCode | default "zh-cn" -}} {{- /* Keywords: front matter tags > site params */ -}} {{- $keywords := $page.Keywords -}} {{- /* Page title */ -}} {{- $title := $page.Title -}} {{- if not $title -}} {{- $title = $site.Title -}} {{- end -}} {{- /* Open Graph */ -}} {{- with $image }} {{- end }} {{- if eq $ogType "article" }} {{- with $page.Params.categories }} {{- end }} {{- range $page.Params.tags }} {{- end }} {{- with $page.Date }} {{- end }} {{- with $page.Lastmod }} {{- end }} {{- with $author }} {{- end }} {{- end }} {{- /* Twitter Card */ -}} {{- with $image }} {{- end }} {{- with $site.Params.twitter }} {{- end }} {{- with $author }} {{- end }} {{- /* JSON-LD structured data */ -}} {{- partial "head/jsonld.html" . }}
